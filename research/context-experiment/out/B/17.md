## 2. `app/src/lib/markdown.ts`

在 `coverMatch` 之后加一行解析（不设默认值，没有就是 `undefined`，代表不属于任何系列）：

```ts
const coverMatch = frontmatter.match(/cover:\s*["']?([^"'\n]+)["']?/);
if (coverMatch) metadata.coverImage = coverMatch[1].trim();

const seriesMatch = frontmatter.match(/series:\s*["']?([^"'\n]+)["']?/);
if (seriesMatch) metadata.series = seriesMatch[1].trim();
```

frontmatter 示例（`content/posts/xxx.md`）：

```md
---

category: "技术"
tags: [AI, 转型]
date: "2026-01-10"

```

## 4. `app/src/lib/utils.ts`

`blogs.ts` 里已经有一份「相对路径补 `BASE_URL` 前缀」的逻辑（`resolveCover`），系列封面需要一模一样的处理，抽出来共用，避免两处各写一份以后行为漂移：

```ts
import { clsx, type ClassValue } from "clsx"
import { twMerge } from "tailwind-merge"

export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs))
}

// 绝对 URL 与以 / 开头的路径原样返回；相对路径补部署 base（如 GitHub Pages 的 /personalweb/ 前缀）
export function resolveAssetPath(path: string): string {
  if (!path || /^(https?:)?\/\//.test(path) || path.startsWith('/')) {
    return path;
  }
  return `${import.meta.env.BASE_URL}${path}`;
}
```

## 5. `app/src/data/series.ts`（新增）

系列的封面/简介无法从文章反推，必须人工维护一份配置，`key` 需与文章 frontmatter 的 `series` 字段完全一致：

```ts
/**
 * [INPUT]: 依赖 @/lib/utils 的 resolveAssetPath 补全部署前缀
 * [OUTPUT]: 对外提供 SeriesInfo/SeriesGroup 类型与 getSeriesInfo 查询
 * [POS]: data/ 层的系列元数据配置，被 data/blogs 的 getSeriesGroups 消费。
 *        key 必须与文章 frontmatter 里的 series 字段完全一致，
 *        否则该系列文章会被分组但拿不到封面（此时 cover 为空字符串，UI 侧需兜住）
 * [PROTOCOL]: 变更时更新此头部，然后检查 CLAUDE.md
 */
import { resolveAssetPath } from '@/lib/utils';
import type { BlogPost } from '@/types/blog';

export interface SeriesInfo {
  name: string;
  cover: string;
  description?: string;
}

export interface SeriesGroup extends SeriesInfo {
  posts: BlogPost[];
}

interface SeriesConfigEntry {
  cover: string;
  description?: string;
}

// cover 填相对路径，图片放在 public/series/ 下
const seriesConfig: Record<string, SeriesConfigEntry> = {
  'AI 转型指南': {
    cover: 'series/ai-transformation.svg',
    description: '从零开始梳理 AI 时代的转型路径',
  },
};

export function getSeriesInfo(name: string): SeriesInfo {
  const config = seriesConfig[name];
  return {
    name,
    cover: config ? resolveAssetPath(config.cover) : '',
    description: config?.description,
  };
}
```

## 6. `app/src/data/blogs.ts`

两处改动：① 复用 `resolveAssetPath` 替换本地的 `resolveCover`；② 新增 `getSeriesGroups`。

```ts
import { resolveAssetPath } from '@/lib/utils';
import { getSeriesInfo, type SeriesGroup } from './series';

export type { SeriesGroup };

// 删除原来本地定义的 resolveCover，改为直接用 resolveAssetPath
const staticBlogPosts: BlogPost[] = Object.entries(localPostModules)
  .map(([path, raw]) => {
    const slug = path.split('/').pop()!.replace(/\.md$/, '');
    const post = { title: slug, excerpt: '', coverImage: '', ...parseBlogMetadata(raw, slug) } as BlogPost;
    return { ...post, coverImage: resolveAssetPath(post.coverImage) };
  })
  .sort((a, b) => b.date.localeCompare(a.date));

// ... 文件末尾追加

/**
 * 按系列分组文章。系列内部按日期升序（第1篇在前），
 * 与列表“最新在前”的顺序相反——系列强调阅读顺序，不是发布时间。
 */
export async function getSeriesGroups(): Promise<SeriesGroup[]> {
  const posts = await getBlogPosts();
  const grouped = new Map<string, BlogPost[]>();

  posts.forEach((post) => {
    if (!post.series) return;
    const list = grouped.get(post.series) ?? [];
    list.push(post);
    grouped.set(post.series, list);
  });

  return Array.from(grouped.entries()).map(([name, seriesPosts]) => ({
    ...getSeriesInfo(name),
    posts: [...seriesPosts].sort((a, b) => a.date.localeCompare(b.date)),
  }));
}
```

## 7. `app/src/pages/Blog.tsx`

导入、状态、数据加载、筛选逻辑、UI 四处改动：

```tsx
import { getBlogPosts, getCategories, getTags, getSeriesGroups, type BlogPost, type SeriesGroup } from '@/data/blogs';

// state 新增
const [seriesGroups, setSeriesGroups] = useState<SeriesGroup[]>([]);
const [activeSeries, setActiveSeries] = useState<string | null>(null);

// loadBlogData 里一并拉取
const [loadedPosts, loadedCategories, loadedTags, loadedSeries] = await Promise.all([
  getBlogPosts(),
  getCategories(),
  getTags(),
  getSeriesGroups(),
]);
setPosts(loadedPosts);
setCategories(loadedCategories);
setTags(loadedTags);
setSeriesGroups(loadedSeries);

// filteredPosts 加入系列筛选
const filteredPosts = posts.filter((post) => {
  const matchesCategory = activeCategory === '全部' || post.category === activeCategory;
  const matchesTag = !activeTag || post.tags.includes(activeTag);
  const matchesSeries = !activeSeries || post.series === activeSeries;
  const matchesSearch =
    !searchQuery ||
    post.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
    post.excerpt.toLowerCase().includes(searchQuery.toLowerCase());
  return matchesCategory && matchesTag && matchesSeries && matchesSearch;
});
```

在 `lg:col-span-2` 主内容列、搜索框之前插入系列分组区块：

```tsx
{seriesGroups.length > 0 && (
  <div className="mb-6">
    <h2 className="text-lg font-medium text-[#1a1a1a] mb-4">系列专栏</h2>
    <div className="grid sm:grid-cols-2 gap-4">
      {seriesGroups.map((series) => (
        <button
          key={series.name}
          onClick={() => setActiveSeries(activeSeries === series.name ? null : series.name)}
          className={`text-left group bg-white rounded-xl overflow-hidden shadow-sm hover:shadow-lg transition-all ${
            activeSeries === series.name ? 'ring-2 ring-[#1a1a1a]' : ''
          }`}
        >
          {series.cover && (
            <div className="aspect-video overflow-hidden">
              <img
                src={series.cover}
                alt={series.name}
                className="w-full h-full object-cover group-hover:scale-105 transition-transform"
              />
            </div>
          )}
          <div className="p-4">
            <h3 className="font-medium text-[#1a1a1a]">{series.name}</h3>
            <p className="text-[#9ca3af] text-sm mt-1">共 {series.posts.length} 篇</p>
          </div>
        </button>
      ))}
    </div>
    {activeSeries && (
      <button
        onClick={() => setActiveSeries(null)}
        className="mt-3 text-sm text-[#6b6b6b] hover:text-[#1a1a1a]"
      >
        清除系列筛选 · {activeSeries} ×
      </button>
    )}
  </div>
)}
```

文章卡片上顺带加个系列徽标（在原有 category badge 旁边）：

## 8. 封面图片本体

放在 `app/public/series/` 下（与现有文章封面 `covers/x.svg` 同级约定），例如 `app/public/series/ai-transformation.svg`，然后在 `series.ts` 的 `seriesConfig` 里填 `cover: 'series/ai-transformation.svg'`。请确认现有 `covers/` 目录的真实位置（构建产物里应能通过 `/covers/xxx.svg` 直接访问），`series/` 目录与其保持同一父目录即可。
