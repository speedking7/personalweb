## 5. `app/src/data/blogs.ts`

把私有的 `resolveCover` 导出（供本文件新函数使用即可，签名不变，只加 `export`）：

```ts
export const resolveCover = (cover: string): string =>
  !cover || /^(https?:)?\/\//.test(cover) || cover.startsWith('/')
    ? cover
    : `${import.meta.env.BASE_URL}${cover}`;
```

顶部新增 import：

```ts
import { seriesCovers } from './series';
```

文件末尾（`clearCache` 附近）新增：

```ts
export interface SeriesGroup {
  name: string;
  cover: string;
  posts: BlogPost[];
}

/**
 * 按 series 字段对文章分组。
 * 组内按日期升序排列（还原系列的阅读顺序），与全站列表默认的降序刻意不同。
 * 封面优先取 seriesCovers 注册表，未登记时兜底用组内第一篇文章的封面。
 */
export async function getSeriesGroups(): Promise<SeriesGroup[]> {
  const posts = await getBlogPosts();
  const grouped = new Map<string, BlogPost[]>();

  posts.forEach((post) => {
    if (!post.series) return;
    const list = grouped.get(post.series) ?? [];
    list.push(post);
    grouped.set(post.series, list);
  });

  return Array.from(grouped.entries()).map(([name, seriesPosts]) => {
    const sorted = [...seriesPosts].sort((a, b) => a.date.localeCompare(b.date));
    return {
      name,
      cover: resolveCover(seriesCovers[name] ?? '') || sorted[0]?.coverImage || '',
      posts: sorted,
    };
  });
}
```

## 使用方式

**写系列文章**：在文章 frontmatter 里加一行

```markdown
---
title: "从 0 到 1：AI 转型第一步"
category: 技术
series: AI 转型指南
date: 2026-08-01
---
```

同系列的多篇文章用相同的 `series` 值即可自动归组，无需改动其它文件。

**配系列封面**：把图片放进 `app/public/series-covers/`（新建这个目录），再去 `app/src/data/series.ts` 里加一行映射：

```ts
'AI 转型指南': 'series-covers/ai-transform.svg',
```

不配也不会出错——`getSeriesGroups` 会自动兜底用该系列第一篇文章的封面。
